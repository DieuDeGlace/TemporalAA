cmake_minimum_required(VERSION 3.20)
project(TemporalAA LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MSVC)
  message(FATAL_ERROR "Use clang-cl in an x64 Visual Studio tools environment")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SMAA/SMAA.hlsl" SMAA_SOURCE)
string(REPLACE "SamplerState LinearSampler { Filter = MIN_MAG_LINEAR_MIP_POINT; AddressU = Clamp; AddressV = Clamp; };" "SamplerState LinearSampler:register(s0);" SMAA_SOURCE "${SMAA_SOURCE}")
string(REPLACE "SamplerState PointSampler { Filter = MIN_MAG_MIP_POINT; AddressU = Clamp; AddressV = Clamp; };" "SamplerState PointSampler:register(s1);" SMAA_SOURCE "${SMAA_SOURCE}")
configure_file(SmaaShader.h.in SmaaShader.h @ONLY)
function(configure_mod target)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cwmods")
  target_link_libraries(${target} PRIVATE d3d11 dxgi d3dcompiler user32 winmm)
endfunction()
add_library(TemporalAA SHARED TemporalAA.cpp)
configure_mod(TemporalAA)
include(CTest)
if(BUILD_TESTING)
 add_executable(TemporalTest Tests/TemporalTest.cpp)
 configure_mod(TemporalTest)
 target_compile_options(TemporalTest PRIVATE /UNDEBUG)
 add_test(NAME TemporalTest COMMAND TemporalTest)
endif()
